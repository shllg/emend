import { createHash } from "node:crypto";
import { mkdir, open, readFile, rename, rm, stat } from "node:fs/promises";
import { dirname, join } from "node:path";

export class BlobStore {
  constructor(private readonly root: string) {}

  private path(ownerId: string, sha256: string): string {
    return join(this.root, ownerId, sha256.slice(0, 2), sha256);
  }

  async has(ownerId: string, sha256: string): Promise<boolean> {
    return stat(this.path(ownerId, sha256)).then(
      (info) => info.isFile(),
      () => false,
    );
  }

  async read(ownerId: string, sha256: string): Promise<Buffer> {
    return readFile(this.path(ownerId, sha256));
  }

  async write(ownerId: string, sha256: string, bytes: Buffer): Promise<void> {
    if (createHash("sha256").update(bytes).digest("hex") !== sha256) throw new Error("Blob hash mismatch");
    const target = this.path(ownerId, sha256);
    await mkdir(dirname(target), { recursive: true });
    const temporary = `${target}.tmp`;
    const handle = await open(temporary, "w", 0o600);
    try {
      await handle.writeFile(bytes);
      await handle.sync();
    } finally {
      await handle.close();
    }
    await rename(temporary, target).catch(async (error) => {
      await rm(temporary, { force: true });
      throw error;
    });
  }
}
