import { and, eq } from "drizzle-orm";
import type { Hono } from "hono";
import { reviews } from "../../db/schema.ts";
import type { AppDeps } from "../app.ts";
import type { AppEnv } from "../auth/principal.ts";
import { ApiError } from "../http/errors.ts";

type Transaction = Parameters<Parameters<AppDeps["db"]["transaction"]>[0]>[0];

/** Moves an open review to a closed state. Retire and approve share it. */
async function closeReview(
  tx: Transaction,
  deps: AppDeps,
  reviewId: string,
  ownerId: string,
  state: "approved" | "settled",
) {
  const [review] = await tx
    .select({ state: reviews.state })
    .from(reviews)
    .where(and(eq(reviews.id, reviewId), eq(reviews.ownerId, ownerId)));
  if (!review) throw new ApiError(404, "unknown_review", "Unknown review");
  if (review.state !== "open") throw new ApiError(409, "invalid_state", "Only an open review can be closed");
  await tx
    .update(reviews)
    .set({
      state,
      snoozedUntil: null,
      updatedAt: deps.now(),
    })
    .where(and(eq(reviews.id, reviewId), eq(reviews.ownerId, ownerId)));
}

export function mountReviewRoutes(app: Hono<AppEnv>, deps: AppDeps) {
  app.post("/api/v1/reviews/:id/retire", async (c) => {
    const { ownerId } = c.get("principal");
    const id = c.req.param("id");
    await deps.db.transaction((tx) => closeReview(tx, deps, id, ownerId, "settled"));
    return c.body(null, 204);
  });

  app.post("/api/v1/reviews/:id/approve", async (c) => {
    const { ownerId } = c.get("principal");
    const id = c.req.param("id");
    await deps.db.transaction((tx) => closeReview(tx, deps, id, ownerId, "approved"));
    return c.body(null, 204);
  });
}
