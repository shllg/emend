import { describe, expect, test } from "vitest";
import { inboxRow } from "../test/rows.ts";
import { sortInbox } from "./inbox-order.ts";

describe("sortInbox", () => {
  test("pinned reviews sort first inside the Open group, newest pin on top", () => {
    const rows = [
      inboxRow({ id: "a", updatedAt: "2026-11-03T10:00:00Z" }),
      inboxRow({ id: "b", updatedAt: "2026-11-01T10:00:00Z", pinnedAt: "2026-11-02T10:00:00Z" }),
      inboxRow({ id: "c", updatedAt: "2026-11-01T09:00:00Z", pinnedAt: "2026-11-03T09:00:00Z" }),
    ];
    expect(sortInbox(rows).map((row) => row.id)).toEqual(["c", "b", "a"]);
  });

  test("a pinned review does not leave the Done group", () => {
    const rows = [
      inboxRow({ id: "a", state: "approved", pinnedAt: "2026-11-03T09:00:00Z" }),
      inboxRow({ id: "b" }),
    ];
    expect(sortInbox(rows).map((row) => row.id)).toEqual(["b", "a"]);
  });
});
